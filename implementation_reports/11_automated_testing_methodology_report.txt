2. TESTING STACK & CONFIGURATION
--------------------------------------------------------------------------------
- Test Runner: Jest ('jest').
- HTTP Endpoint Testing: Supertest ('supertest').
- Database Strategy: Tests execute against test database or isolated test
  transactions that rollback upon completion.

Installation:
npm install --save-dev jest supertest

Add to 'package.json':
"scripts": {
    "test": "jest --runInBand --detectOpenHandles --forceExit"
}

4. SAMPLE EXECUTABLE TEST SUITES
--------------------------------------------------------------------------------
A. 'tests/health.test.js':
   const request = require('supertest');
   const app = require('../index'); // Export app from index.js

   describe('GET /health', () => {
       it('should return 200 and database healthy status', async () => {
           const res = await request(app).get('/health');
           expect(res.statusCode).toEqual(200);
           expect(res.body).toHaveProperty('status', 'healthy');
           expect(res.body).toHaveProperty('database', 'connected');
       });
   });

B. 'tests/menu.test.js':
   describe('Menu Endpoints', () => {
       it('GET /api/menu should return a non-empty array of drinks', async () => {
           const res = await request(app).get('/api/menu');
           expect(res.statusCode).toEqual(200);
           expect(Array.isArray(res.body)).toBe(true);
           expect(res.body.length).toBeGreaterThan(0);
           expect(res.body[0]).toHaveProperty('name');
           expect(res.body[0]).toHaveProperty('price');
       });

       it('PUT /api/menu/:id should reject negative prices', async () => {
           const res = await request(app)
               .put('/api/menu/1')
               .send({ price: -5.00 });
           expect(res.statusCode).toBeGreaterThanOrEqual(400);
       });
   });

C. 'tests/orders.test.js':
   describe('POST /api/orders', () => {
       it('should reject orders with an empty item list', async () => {
           const res = await request(app)
               .post('/api/orders')
               .send({ customer_name: 'Test', items: [] });
           expect(res.statusCode).toEqual(400);
       });
   });
